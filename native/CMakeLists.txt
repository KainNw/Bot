cmake_minimum_required(VERSION 3.20)
project(AtlanticaBridge LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "The client bridge requires 32-bit MSVC on Windows")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_library(AtlanticaHook SHARED hook.cpp hook.def)
target_link_libraries(AtlanticaHook PRIVATE user32)
add_executable(AtlanticaBridge bridge.cpp)
target_link_libraries(AtlanticaBridge PRIVATE user32 advapi32)
target_compile_options(AtlanticaHook PRIVATE /W4 /WX /EHsc)
target_compile_options(AtlanticaBridge PRIVATE /W4 /WX /EHsc)
enable_testing()
add_test(NAME own_window_hook_and_build_guard COMMAND AtlanticaBridge --self-test)
