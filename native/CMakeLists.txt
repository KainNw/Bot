cmake_minimum_required(VERSION 3.20)
project(AtlanticaBridge LANGUAGES C CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "The client bridge requires 32-bit MSVC on Windows")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_library(MinHook STATIC third_party/minhook/src/buffer.c third_party/minhook/src/hook.c
  third_party/minhook/src/trampoline.c third_party/minhook/src/hde/hde32.c)
target_compile_options(MinHook PRIVATE /W0)
add_library(ClockScaler STATIC clock_scaler.cpp)
target_compile_options(ClockScaler PRIVATE /W4 /WX /EHsc)
target_link_libraries(ClockScaler PUBLIC MinHook)
add_library(AtlanticaHook SHARED hook.cpp clock_control.cpp auto_life.cpp hook.def)
target_link_libraries(AtlanticaHook PRIVATE user32 ClockScaler)
add_executable(AtlanticaBridge bridge.cpp)
target_link_libraries(AtlanticaBridge PRIVATE user32 advapi32)
target_compile_options(AtlanticaHook PRIVATE /W4 /WX /EHsc)
target_compile_options(AtlanticaBridge PRIVATE /W4 /WX /EHsc)
enable_testing()
add_test(NAME own_window_hook_and_build_guard COMMAND AtlanticaBridge --self-test)
add_executable(ClockSelfTest clock_self_test.cpp)
target_compile_options(ClockSelfTest PRIVATE /W4 /WX /EHsc)
target_link_libraries(ClockSelfTest PRIVATE ClockScaler winmm)
add_test(NAME own_process_clock_scaling COMMAND ClockSelfTest)
add_executable(ClockControllerSelfTest clock_controller_self_test.cpp clock_control.cpp)
target_compile_definitions(ClockControllerSelfTest PRIVATE ATLANTICA_CLOCK_TEST_LIMIT=300)
target_compile_options(ClockControllerSelfTest PRIVATE /W4 /WX /EHsc)
target_link_libraries(ClockControllerSelfTest PRIVATE ClockScaler user32)
add_test(NAME own_fixture_clock_lifecycle COMMAND ClockControllerSelfTest)
add_executable(GeneralClockControllerSelfTest general_clock_controller_self_test.cpp clock_control.cpp)
target_compile_definitions(GeneralClockControllerSelfTest PRIVATE ATLANTICA_CLOCK_TEST_LIMIT=300)
target_compile_options(GeneralClockControllerSelfTest PRIVATE /W4 /WX /EHsc)
target_link_libraries(GeneralClockControllerSelfTest PRIVATE ClockScaler user32)
add_test(NAME own_fixture_general_clock_lifecycle COMMAND GeneralClockControllerSelfTest)
add_executable(AutoLifeSelfTest auto_life_self_test.cpp)
target_compile_options(AutoLifeSelfTest PRIVATE /W4 /WX /EHsc)
target_link_libraries(AutoLifeSelfTest PRIVATE user32)
add_test(NAME own_window_auto_life_input COMMAND AutoLifeSelfTest)
set_tests_properties(own_window_hook_and_build_guard own_process_clock_scaling
  own_fixture_clock_lifecycle own_fixture_general_clock_lifecycle own_window_auto_life_input PROPERTIES TIMEOUT 30)
